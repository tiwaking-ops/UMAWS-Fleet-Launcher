const $ = id => document.getElementById(id);

/* ---------- helpers ---------- */
function esc(s) {
  return String(s ?? "").replace(/[&<>"]/g, c => (
    {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c]));
}
async function api(path, opts) {
  const r = await fetch(path, opts);
  const t = await r.text();
  try { return JSON.parse(t); } catch { return { error: t }; }
}
function post(path, body) {
  const init = { method: "POST", headers: {"Content-Type": "application/json"} };
  if (body !== undefined) init.body = JSON.stringify(body);
  return api(path, init);
}
function show(name) {
  document.querySelectorAll(".tab").forEach(t =>
    t.classList.toggle("active", t.id === "tab-" + name));
  document.querySelectorAll("nav button").forEach(b =>
    b.classList.toggle("active", b.dataset.tab === name));
  if (name === "launcher") loadLauncher();
  if (name === "tasks") loadTasks();
  if (name === "queue") loadQueue();
  if (name === "overseer") loadOverseer();
  if (name === "metrics") loadMetrics();
}

/* ---------- live stream ---------- */
function connect() {
  const es = new EventSource("/api/events");
  es.onmessage = ev => {
    const d = JSON.parse(ev.data);
    updateHeader(d);
    updateSlots(d.slots || []);
    updateProfiles(d.fleet || {});
    updateProject(d.status || {});
  };
  es.onerror = () => { setTimeout(connect, 3000); };
}
function updateHeader(d) {
  const st = d.status || {};
  $("pill-state").textContent = "state: " + (st.state || "?");
  $("pill-slots").textContent = (d.slots || []).length + " slots";
  const p = $("pill-dispatch");
  p.textContent = d.fleet && d.fleet.dispatch_paused ? "dispatch PAUSED" : "dispatch ok";
  p.className = "pill " + (d.fleet && d.fleet.dispatch_paused ? "bad" : "ok");
}
function updateProject(st) {
  $("project-kpis").innerHTML = [
    ["state", st.state], ["tasks", st.tasks_done + "/" + st.tasks],
    ["goals", st.goals_done + "/" + st.goals_total],
    ["pdd", st.pdd ? "yes" : "no"],
  ].map(([k, v]) =>
    `<div class="kpi"><b>${esc(v)}</b><span>${esc(k)}</span></div>`).join("");
  $("project-detail").textContent =
    `brief: ${st.brief || "-"}   report: ${st.report || "-"}`;
}
function updateSlots(slots) {
  if (!slots.length) { $("slots").innerHTML = '<p class="muted">no slots</p>'; return; }
  $("slots").innerHTML = `<table><tr><th>task</th><th>profile</th><th>pid</th>
    <th>runtime</th><th>last output</th><th></th></tr>` +
    slots.map(s => `<tr>
      <td class="mono">${esc(s.task)}</td>
      <td class="mono">${esc(s.profile)}${s.degraded ? " <span class='pill warn'>degraded</span>" : ""}</td>
      <td class="mono">${esc(s.pid)}</td>
      <td class="mono">${s.runtime_s}s</td>
      <td class="mono">${s.last_output_age_s}s</td>
      <td><button class="act" onclick="tail('${esc(s.task)}')">log</button></td>
    </tr>`).join("") + "</table>";
}
function updateProfiles(fleet) {
  const rows = fleet.profiles || [];
  $("profiles").innerHTML = `<table><tr><th>id</th><th>state</th><th>run/max</th>
    <th>cooldown</th><th>last failure</th><th></th></tr>` +
    rows.map(p => `<tr>
      <td class="mono">${esc(p.id)}</td>
      <td>${esc(p.state)}</td>
      <td class="mono">${p.running}/${p.max_concurrent}</td>
      <td class="mono">${p.cooldown_remaining_s}s</td>
      <td class="mono">${esc(p.last_failure_class || "-")}</td>
      <td>
        <button class="act" onclick="fleetAct('enable','${esc(p.id)}')">on</button>
        <button class="act" onclick="fleetAct('disable','${esc(p.id)}')">off</button>
        <button class="act" onclick="fleetAct('cool','${esc(p.id)}')">cool</button>
      </td></tr>`).join("") + "</table>";
}

/* ---------- actions ---------- */
async function projectInit() {
  const brief = $("brief").value;
  const r = await post("/api/project/init?brief=" + encodeURIComponent(brief));
  log("project init: " + JSON.stringify(r));
}
async function projectAction(a) {
  const r = await post("/api/project/" + a);
  log("project " + a + ": " + JSON.stringify(r));
}
async function closeCheck() {
  const r = await post("/api/project/close-check");
  $("launcher-log").textContent = JSON.stringify(r, null, 2);
  log("close-check: " + (r.ok ? "green" : "not green"));
}
async function fleetAct(a, id) {
  let url = `/api/fleet/${a}?profile=${encodeURIComponent(id)}`;
  if (a === "cool") url += "&minutes=15";
  await post(url);
}
async function tail(task) {
  const r = await api(`/api/agent/${task}/log`);
  $("launcher-log").textContent = (r.log || []).join("\n");
  show("launcher");
}
async function stopAll() {
  const r = await post("/api/stop-all");
  log("stop all: " + JSON.stringify(r));
}
async function killAll() {
  const r = await post("/api/kill-all");
  log("kill all: " + JSON.stringify(r));
}
async function resume() {

}

  const tasks = await api("/api/tasks");
  $("tasks").innerHTML = tasks.length ? `<table><tr><th>id</th><th>state</th>
    <th>type</th><th>title</th><th>goal</th><th>attempts</th><th>scope</th></tr>` +
    tasks.map(t => `<tr>
      <td class="mono">${esc(t.id)}</td>
      <td class="s-${esc(t.state)}">${esc(t.state)}</td>
      <td>${esc(t.type)}</td>
      <td>${esc(t.title)}</td>
      <td class="mono">${esc(t.goal_ref)}</td>
      <td class="mono">${t.attempt}/${esc(t.attempts.max)}</td>
      <td class="mono small">${esc((t.scope.write || []).join(", "))}</td>
    </tr>`).join("") + "</table>" : '<p class="muted">no tasks</p>';

  const items = await api("/api/queue");
  const open = items.filter(i => i.state === "open");
  $("queue").innerHTML = open.length ? `<table><tr><th>id</th><th>kind</th>
    <th>topic</th><th>options</th><th>recommended</th><th>safe default</th><th></th></tr>` +
    open.map(i => `<tr>
      <td class="mono">${esc(i.id)}</td>
      <td>${i.class_k ? "<span class='pill warn'>class K</span>" : "normal"}</td>
      <td>${esc(i.topic)}</td>
      <td class="small">${esc((i.options || []).join(", "))}</td>
      <td class="small">${esc(i.recommended)}</td>
      <td class="small">${esc(i.safe_default)}</td>
      <td><button class="act" onclick="decide('${esc(i.id)}','${esc(i.recommended || (i.options||[])[0] || "")}')">decide</button></td>
    </tr>`).join("") + "</table>" : '<p class="muted">nothing waiting</p>';
}
async function decide(id, choice) {
  await post(`/api/queue/decide?item_id=${encodeURIComponent(id)}&choice=${encodeURIComponent(choice)}`);
  loadQueue();
}
async function loadOverseer() {
  const r = await api("/api/overseer");
  const s = r.status || {};
  $("overseer").innerHTML = `<div class="kpis">
    <div class="kpi"><b>${esc(s.session || "-")}</b><span>session</span></div>

    <div class="kpi"><b>${s.wakes || 0}</b><span>wakes total</span></div>
  </div><p class="small muted">profiles: ${esc((s.profiles || []).join(", ") || "none configured")}</p>`;
  $("wake-log").textContent = (s.recent || []).map(w =>

  ).join("\n") || "(no wakes yet)";
}

  $("metrics").innerHTML = [

    ["wakes", m.wakes],
  ].map(([k, v]) => `<div class="kpi"><b>${esc(v)}</b><span>${esc(k)}</span></div>`).join("");
  $("metric-profiles").innerHTML = `<table><tr><th>profile</th><th>runs</th>
    <th>failures</th><th>last failure</th><th>state</th></tr>` +
    (m.per_profile || []).map(p => `<tr><td class="mono">${esc(p.id)}</td>
      <td class="mono">${p.runs}</td><td class="mono">${p.failures}</td>
      <td class="mono">${esc(p.last_failure_class || "-")}</td>
      <td>${esc(p.state)}</td></tr>`).join("") + "</table>";
}

/* ---------- manual launcher ---------- */
async function loadLauncher() {
  const d = await api("/api/launcher");
  launcherEntries = d.entries || [];
  $("l-delay").value = (d.settings && d.settings.delay_seconds) || 1.0;
  $("l-keep").checked = !!(d.settings && d.settings.keep_open);
  const groups = ["All groups", ...new Set(launcherEntries.map(e => e.group).filter(Boolean))].sort();
  const sel = $("l-group");
  const current = sel.value;
  sel.innerHTML = groups.map(g => `<option>${esc(g)}</option>`).join("");
  sel.value = groups.includes(current) ? current : "All groups";
  renderLauncher();
}
function renderLauncher() {
  const q = $("l-search").value.toLowerCase();
  const g = $("l-group").value;
  const rows = launcherEntries.filter(e =>
    (g === "All groups" || e.group === g) &&
    (!q || [e.name, e.group, e.prefix, e.cwd, e.prompt].join(" ").toLowerCase().includes(q)));
  $("launcher-table").innerHTML = `<table><tr><th></th><th>name</th><th>group</th>
    <th>prefix</th><th>cwd</th><th></th></tr>` + rows.map(e => `<tr>
    <td><input type="checkbox" style="width:auto" ${checked.has(e.id) ? "checked" : ""}
      onchange="toggle('${esc(e.id)}', this.checked)"></td>
    <td>${esc(e.name)}</td><td>${esc(e.group)}</td>
    <td class="mono small">${esc(e.prefix)}</td>
    <td class="mono small">${esc(e.cwd)}</td>
    <td><button class="act" onclick="editEntry('${esc(e.id)}')">edit</button></td>
  </tr>`).join("") + "</table>";
}
function toggle(id, on) { on ? checked.add(id) : checked.delete(id); }
function lSelectVisible() {
  document.querySelectorAll("#launcher-table input[type=checkbox]").forEach(
    c => { if (c.checked) checked.add(c.closest("tr").querySelector("button").getAttribute("onclick").match(/'([^']+)'/)[1]); });
  renderLauncher();
}
function lClearVisible() {
  document.querySelectorAll("#launcher-table input[type=checkbox]").forEach(
    c => { if (c.checked) checked.delete(c.closest("tr").querySelector("button").getAttribute("onclick").match(/'([^']+)'/)[1]); });
  renderLauncher();
}
function lClearAll() { checked.clear(); renderLauncher(); }
function lNew() {
  const e = { id: "new-" + Date.now(), name: "", group: "", prefix: "",
              prompt: "", suffix: "", cwd: "" };
  launcherEntries.push(e);
  editEntry(e.id);
}
function lDuplicate() {
  const copies = [];
  launcherEntries.forEach(e => { if (checked.has(e.id)) {
    const c = { ...e, id: "dup-" + Date.now() + Math.random(), name: e.name + " (copy)" };
    launcherEntries.push(c); copies.push(c.id); } });
  if (copies.length) { copies.forEach(id => checked.add(id)); renderLauncher(); }
}
function lDelete() {
  launcherEntries = launcherEntries.filter(e => !checked.has(e.id));
  checked.clear(); renderLauncher(); saveLauncher();
}
function editEntry(id) {
  const e = launcherEntries.find(x => x.id === id);
  if (!e) return;
  const p = prompt("name", e.name), g = prompt("group", e.group),
        pre = prompt("command prefix", e.prefix), pr = prompt("prompt", e.prompt),
        su = prompt("extra args", e.suffix), cw = prompt("working dir", e.cwd);
  if (p === null) return;
  Object.assign(e, { name: p, group: g, prefix: pre, prompt: pr, suffix: su, cwd: cw });
  renderLauncher(); saveLauncher();
}
function saveLauncher() {
  post("/api/launcher/save", { entries: launcherEntries,
    settings: { delay_seconds: parseFloat($("l-delay").value) || 1.0,
                keep_open: $("l-keep").checked } });
}
function lImport() {
  const path = prompt("path to a text file of command lines");
  if (!path) return;
  post("/api/launcher/import?path=" + encodeURIComponent(path))
    .then(r => { log("import: " + JSON.stringify(r)); loadLauncher(); });
